💻 Cara Pakai

// JavaScript
fetch('https://api.kamu.com/api/islami/jadwal-sholat?kota=Jakarta&apikey=KEY_KAMU')
  .then(r => r.json())
  .then(d => console.log(d.jadwal));

# Python
import requests
r = requests.get('https://api.kamu.com/api/crypto/harga',
    params={'coin': 'bitcoin', 'apikey': 'KEY_KAMU'})
print(r.json())

# cURL
curl "https://api.kamu.com/api/cuaca?kota=jakarta&apikey=KEY_KAMU"
